351  {
352  // The search backtracks over guesses of a linear map, which finishes quickly for anything that looks
353  // like a real S-box but degenerates on tables that are close to linear, as those have too many linear
354  // self-equivalences to enumerate. Such a table cannot be a real S-box, so give up on it instead:
355  // R_S_best then holds the best representative found so far, which still belongs to the equivalence
356  // class of S, so a truncated search can only miss a match in the database, never invent one.
357  if (budget == 0)
358  {
359  return false;
360  }
361  budget--;
362 
363  std::vector<u8> A(state.A);
364  std::vector<u8> B(state.B);
365  std::vector<u8> R_S(state.R_S);

368  smallset_t D_B = (state.D_B);
369  smallset_t C_A = (state.C_A);
370  smallset_t C_B = (state.C_B);
371  smallset_t N_A = (state.N_A);
372  smallset_t N_B = (state.N_B);
373  smallset_t U_A = (state.U_A);
374  smallset_t U_B = (state.U_B);
375 
376  while (!smallset_is_empty(N_A))
377  {
378  u8 x = smallset_least_element(N_A);
379  u8 y = smallset_least_element(U_B);
380 
381  B[y] = S[A[x]];
382  if (!update_linear(B, y, len))
383  return false;
384  smallset_t D_B_new = smallset_shift(D_B, y);
385  D_B = smallset_union(D_B, D_B_new);

550  std::vector<u8> compute_linear_representative_bounded(const std::vector<u8>& sbox, u64& budget)
551  {
552  u32 len = sbox.size();
553 
554  // variable for current best candidate
555  std::vector<u8> R_S_best(len, 0);
556 
557  // invert sbox
558  std::vector<u8> S_inv(len, 0);
559  for (u32 x = 0; x < len; x++)
560  {
561  u8 y = sbox[x];
562  S_inv[y] = x;
563  }
564 
565  // init the state of the algorithm
566  state_t state;
567  state.A = std::vector<u8>(len, 0);
568  state.B = std::vector<u8>(len, 0);
569  state.R_S = std::vector<u8>(len, 0);
570 
571  state.D_A = smallset_add_element(smallset_init_empty(), 0);
572  state.D_B = smallset_add_element(smallset_init_empty(), 0);
573 
574  state.C_A = smallset_init_empty();
575  state.C_B = smallset_init_empty();
576 
577  state.N_A = smallset_add_element(smallset_init_empty(), 0);
578  state.N_B = smallset_add_element(smallset_init_empty(), 0);
579 
580  state.U_A = smallset_setminus(smallset_init_full(len), state.D_A, len);
581  state.U_B = smallset_setminus(smallset_init_full(len), state.D_A, len);
582 
583  // init in special case S[0] == 0
584  if (sbox[0] == 0)
585  {
586  state.C_A = smallset_add_element(smallset_init_empty(), 0);
587  state.C_B = smallset_add_element(smallset_init_empty(), 0);
588 
589  state.N_A = smallset_init_empty();
590  state.N_B = smallset_init_empty();
591  }
592 
593  // compute linear representative recursively
594  subroutine(sbox, S_inv, state, R_S_best, len, budget);
595 
596  return R_S_best;
