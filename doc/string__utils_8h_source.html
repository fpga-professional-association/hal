52  inline std::string trim(const std::string& s)
53  {
54  size_t p0 = 0;
55  size_t p1 = s.size();
56  while (p0 < p1 && (unsigned char)s[p0] <= ' ')
57  {
58  ++p0;
59  }
60  while (p1 > p0 && (unsigned char)s[p1 - 1] <= ' ')
61  {
62  --p1;
63  }
64  return s.substr(p0, p1 - p0);
65  }

207  inline std::string number(int64_t value, int base = 10)
208  {
209  if (base == 10)
210  {
211  return std::to_string(value);
212  }
213  bool negative = value < 0;
214  uint64_t absval = negative ? (uint64_t)(-value) : (uint64_t)value;
215  std::string digits;
216  if (!absval)
217  {
218  digits = "0";
219  }
220  while (absval)
221  {
222  int dig = (int)(absval % (uint64_t)base);
223  digits += (char)(dig < 10 ? ('0' + dig) : ('a' + dig - 10));
224  absval /= (uint64_t)base;
225  }
226  std::reverse(digits.begin(), digits.end());
227  return negative ? ("-" + digits) : digits;
228  }

234  inline std::string number_padded(uint64_t value, int width, int base)
235  {
236  std::string digits;

239  digits = "0";
240  }

242  {
243  int dig = (int)(value % (uint64_t)base);
244  digits += (char)(dig < 10 ? ('0' + dig) : ('a' + dig - 10));
245  value /= (uint64_t)base;

247  std::reverse(digits.begin(), digits.end());
