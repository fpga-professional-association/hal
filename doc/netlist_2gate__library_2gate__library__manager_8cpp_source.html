146  std::vector<std::filesystem::path> resolve_search_list(const std::vector<std::string>& entries)
147  {
148  std::vector<std::filesystem::path> resolved;
149  std::set<std::string> already_resolved;
150 
151  auto append = [&resolved, &already_resolved](const std::filesystem::path& path) {
152  const std::filesystem::path absolute_path = std::filesystem::absolute(path);
153  if (already_resolved.insert(absolute_path.string()).second)
154  {
155  resolved.push_back(absolute_path);
156  }
157  else
158  {
159  log_info("gate_library_manager", "gate library file '{}' is listed more than once, ignoring the later occurrence.", absolute_path.string());
160  }
161  };
162 
163  for (const auto& entry : entries)
164  {
165  if (entry.empty())
166  {
167  continue;
168  }
169 
170  std::error_code ec;
171  const std::filesystem::path entry_path(entry);
172 
173  if (std::filesystem::is_directory(entry_path, ec))
174  {
175  // sort so that the resulting order does not depend on the order the file system reports
176  std::vector<std::filesystem::path> in_directory;
177  for (const auto& lib_path : utils::RecursiveDirectoryRange(entry_path))
178  {
179  if (gate_library_parser_manager::can_parse(lib_path.path()))
180  {
181  in_directory.push_back(lib_path.path());
182  }
183  }
184  std::sort(in_directory.begin(), in_directory.end());
185 
186  if (in_directory.empty())
187  {
188  log_warning("gate_library_manager", "gate library search list entry '{}' is a directory that does not contain any gate library files.", entry);
189  continue;
190  }
191 
192  for (const auto& lib_path : in_directory)
193  {
194  append(lib_path);
195  }
196  }
197  else if (std::filesystem::exists(entry_path, ec))
198  {
199  append(entry_path);
200  }
201  else
202  {
203  // not a path that exists, so search the standard gate library directories for the file name
204  const auto stripped_name = entry_path.filename();
205  const auto lib_path = utils::get_file(stripped_name, utils::get_gate_library_directories());
206  if (lib_path.empty())
207  {
208  log_error("gate_library_manager", "could not resolve gate library search list entry '{}': no such file, directory, or gate library in the default directories.", entry);
209  continue;
210  }
211  append(lib_path);
212  }
213  }
214

218  GateLibrary* load_multiple(const std::vector<std::filesystem::path>& file_paths, const std::string& name, bool reload)
219  {
220  if (file_paths.empty())
221  {
222  log_error("gate_library_manager", "could not load gate libraries: no gate library file given.");
223  return nullptr;
224  }
225 
226  // a single library needs no composition, so keep it addressable by its own file
227  if (file_paths.size() == 1)
228  {
229  return load(file_paths.front(), reload);
230  }
231 
232  std::vector<std::filesystem::path> absolute_paths;
233  for (const auto& file_path : file_paths)
234  {
235  if (!std::filesystem::exists(file_path))
236  {
237  log_error("gate_library_manager", "gate library file '{}' does not exist.", file_path.string());
238  return nullptr;
239  }
240  absolute_paths.push_back(std::filesystem::absolute(file_path));
241  }
242 
243  const std::filesystem::path key = get_composite_key(absolute_paths);
244 
245  if (!reload)
246  {
247  if (auto it = m_gate_libraries.find(key); it != m_gate_libraries.end())
248  {
249  log_info("gate_library_manager", "the gate libraries '{}' are already loaded as a composite gate library.", key.string());
250  return it->second.get();
251  }
252  }
253 
254  // parse fresh copies: the gate types are moved into the composite library, which must not tear apart a
255  // library that is already loaded and possibly in use by a netlist
256  std::vector<std::unique_ptr<GateLibrary>> parsed_libraries;
257  for (const auto& file_path : absolute_paths)
258  {
259  std::unique_ptr<GateLibrary> parsed = gate_library_parser_manager::parse(file_path);
260  if (parsed == nullptr)
261  {
262  log_error("gate_library_manager", "could not load gate library '{}' as part of a composite gate library.", file_path.string());
263  return nullptr;
264  }
265  parsed_libraries.push_back(std::move(parsed));
266  }
267 
268  std::string composite_name = name;
269  if (composite_name.empty())
270  {
271  for (u32 i = 0; i < parsed_libraries.size(); i++)
272  {
273  composite_name += (i == 0 ? "" : "+") + parsed_libraries.at(i)->get_name();
274  }
275  }
276 
277  // start out without a path so that the source paths of the absorbed libraries are the only provenance
278  std::shared_ptr<GateLibrary> composite = std::make_shared<GateLibrary>(std::filesystem::path(), composite_name);
279 
280  for (u32 i = 0; i < parsed_libraries.size(); i++)
281  {
282  if (auto res = composite->absorb(parsed_libraries.at(i).get()); res.is_error())
283  {
284  log_error("gate_library_manager", "could not absorb gate library '{}' into composite gate library:\n{}", absolute_paths.at(i).string(), res.get_error().get());
285  return nullptr;
286  }
287  }
288 
289  composite->set_path(key);
290 
291  if (auto res = prepare_library(composite); res.is_error())
292  {
293  log_error("gate_library_manager", "error encountered while loading composite gate library:\n{}", res.get_error().get());
294  return nullptr;
295  }
296 
297  log_info("gate_library_manager", "loaded composite gate library '{}' from {} gate library files.", composite->get_name(), absolute_paths.size());
298 
299  GateLibrary* res = composite.get();
300  m_gate_libraries[key] = std::move(composite);
301  return res;
302  }
