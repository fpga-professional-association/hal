238  Result<GateType*> GateLibrary::create_black_box_gate_type(const std::string& name, const std::vector<std::pair<std::string, u32>>& ports)
239  {
240  if (m_gate_type_map.find(name) != m_gate_type_map.end())
241  {
242  return ERR("could not create black box gate type '" + name + "': a gate type with the same name already exists within gate library '" + m_name + "'");
243  }
244 
245  // no properties: nothing is known about the internals of a black box, not even whether it is combinational
246  GateType* gt = create_gate_type(name, {});
247  if (gt == nullptr)
248  {
249  return ERR("could not create black box gate type '" + name + "' within gate library '" + m_name + "': failed to create gate type");
250  }
251 
252  for (const auto& [port_name, width] : ports)
253  {
254  if (width == 0)
255  {
256  continue;
257  }
258 
259  // the direction of a port cannot be recovered from a netlist instantiation, hence 'inout'
260  if (width == 1)
261  {
262  if (auto res = gt->create_pin(port_name, PinDirection::inout); res.is_error())
263  {
264  return ERR_APPEND(res.get_error(), "could not create black box gate type '" + name + "' within gate library '" + m_name + "': failed to create pin '" + port_name + "'");
265  }
266  continue;
267  }
268 
269  std::vector<GatePin*> pins;
270  for (u32 i = width; i > 0; i--)
271  {
272  const std::string pin_name = port_name + "(" + std::to_string(i - 1) + ")";
273  if (auto res = gt->create_pin(pin_name, PinDirection::inout, PinType::none, false); res.is_error())
274  {
275  return ERR_APPEND(res.get_error(), "could not create black box gate type '" + name + "' within gate library '" + m_name + "': failed to create pin '" + pin_name + "'");
276  }
277  else
278  {
279  pins.push_back(res.get());
280  }
281  }
282 
283  // a descending group of pins given from the most significant bit down is what a bus of a gate library
284  // parsed from a file looks like, and what the netlist parsers expect when they map a bus connection
285  if (auto res = gt->create_pin_group(port_name, pins, PinDirection::inout, PinType::none, false, static_cast<i32>(width - 1)); res.is_error())
286  {
287  return ERR_APPEND(res.get_error(), "could not create black box gate type '" + name + "' within gate library '" + m_name + "': failed to create pin group '" + port_name + "'");
288  }
289  }
290 
291  m_black_box_gate_types[name] = gt;
292  return OK(gt);
293  }
